## Raspberry Pi configuration documentation:
## https://www.raspberrypi.com/documentation/computers/config_txt.html

[pi0]
## Raspberry Pi Zero, Zero W and Zero 2 W
gpu_mem=128

[pi02]
## Raspberry Pi Zero 2 W
dtoverlay=vc4-kms-v3d

[pi2]
gpu_mem=256

[pi3]
## Raspberry Pi 3, 3+, 3A+, CM3 and CM3+
gpu_mem=128

## Legacy touchscreen support.
## Keep only if required by a directly connected Raspberry Pi/BTT display.
dtoverlay=rpi-ft5406

[pi4]
## Applies to Raspberry Pi 4, Pi 400, CM4 and CM4S.
##
## Trixie uses the full KMS display stack.
dtoverlay=vc4-kms-v3d

## gpu_mem has little benefit with the KMS driver.
## Keep a conservative value for compatibility.
gpu_mem=128

[cm4]
## CM4 also reads the [pi4] section above.
##
## Enable USB host mode only when the carrier board requires DWC2.
## Do not enable this blindly on every CM4 carrier.
dtoverlay=dwc2,dr_mode=host

[pi5]
## Applies to Raspberry Pi 5, Pi 500 and CM5.
##
## Enable the full KMS graphics driver.
dtoverlay=vc4-kms-v3d

## Do not use vc4-fkms-v3d on Pi 5.
## Do not set gpu_mem=256; Pi 5 graphics memory is dynamically managed.

[cm5]
## CM5 already reads the [pi5] section.
##
## Add carrier-board-specific settings here when required.
## Example for USB host mode:
# dtoverlay=dwc2,dr_mode=host

[all]

## Enable SPI for Input Shaper.
##
## Warning: SPI may conflict with HyperPixel displays because some models
## use the same GPIO pins.
dtparam=spi=on

## Enable hardware UART.
enable_uart=1

## Disable onboard Bluetooth so the primary UART is available.
dtoverlay=disable-bt

## Enable I2C for Klipper host MCU and accelerometers.
dtparam=i2c_arm=on

## Optional higher I2C clock for compatible accelerometers:
# dtparam=i2c_arm=on,i2c_arm_baudrate=400000

## Trixie images for Pi 4, CM4 and Pi 5 should already use a 64-bit kernel.
## Leave this commented unless building an image that requires it explicitly.
# arm_64bit=1
